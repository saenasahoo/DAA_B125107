Q5: Lexicographically Smallest Connected Component Permutation
ALGORITHM FIND(x)
    while parent[x] ≠ x do
        parent[x] ← parent[parent[x]]
        x ← parent[x]
    return x

ALGORITHM UNION(a, b)
    a ← FIND(a);  b ← FIND(b)
    if a = b then return
    if size[a] < size[b] then swap(a, b)
    parent[b] ← a
    size[a]   ← size[a] + size[b]

ALGORITHM SMALLEST-PERMUTATION(n, A[1..n], m swaps)
Input : permutation A of 1..n, m allowed swaps (u, v) on positions
Output: lexicographically smallest reachable permutation
    // Step 1: record where each value sits; build components
    for i ← 1 to n do
        parent[i] ← i;  size[i] ← 1
        pos[A[i]] ← i
    for each swap (u, v) do UNION(u, v)

    // Step 2: give every component a block in the array vals[]
    run ← 0
    for i ← 1 to n do
        fill[i] ← 0;  ptr[i] ← 0
        if FIND(i) = i then
            off[i] ← run
            run ← run + size[i]

    // Step 3: scan values in ascending order → each block ends up sorted
    for v ← 1 to n do
        r ← FIND(pos[v])
        vals[off[r] + fill[r]] ← v
        fill[r] ← fill[r] + 1

    // Step 4: scan positions in ascending order → smallest unused value of its component
    for i ← 1 to n do
        r ← FIND(i)
        result[i] ← vals[off[r] + ptr[r]]
        ptr[r] ← ptr[r] + 1

    return result

Time : O((n+m)·α(n))        Space: O(n)
