Q1: Online Bipartiteness
ALGORITHM FIND-PARITY(x)
// Returns (root of x, parity of x relative to that root)
    if parent[x] = x then
        return (x, 0)
    (r, p) ← FIND-PARITY(parent[x])
    parity[x] ← parity[x] XOR p          // path compression: x now points to root
    parent[x] ← r
    return (r, parity[x])

ALGORITHM ONLINE-BIPARTITE(V, E)
Input : V vertices, E edges arriving one by one
Output: for each edge, "added" or "violates bipartiteness (skipped)"
    for x ← 0 to V-1 do
        parent[x] ← x
        size[x]   ← 1
        parity[x] ← 0
    for i ← 1 to E do
        read edge (u, v)
        (ru, pu) ← FIND-PARITY(u)
        (rv, pv) ← FIND-PARITY(v)
        if ru = rv then                      // same component
            if pu = pv then                  // same colour → odd cycle
                print "Edge (u,v) violates bipartiteness, skipped"
            else
                print "Edge (u,v) added"
        else                                 // different components → merge
            if size[ru] < size[rv] then swap(ru, rv)
            parent[rv] ← ru
            parity[rv] ← pu XOR pv XOR 1     // puts u and v on opposite sides
            size[ru]   ← size[ru] + size[rv]
            print "Edge (u,v) added"

Time : O(V + E·α(V))        Space: O(V)
