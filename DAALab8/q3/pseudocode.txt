for i ← 0..m: L[i][0] ← 0
for j ← 0..n: L[0][j] ← 0
for i ← 1..m:
    for j ← 1..n:
        if X[i] = Y[j]: L[i][j] ← L[i−1][j−1] + 1
        else: L[i][j] ← max(L[i−1][j], L[i][j−1])
// reconstruct
i ← m, j ← n, S ← empty
while i > 0 and j > 0:
    if X[i] = Y[j]: prepend X[i] to S; i--, j--
    else if L[i−1][j] ≥ L[i][j−1]: i--
    else: j--
return L[m][n], S
