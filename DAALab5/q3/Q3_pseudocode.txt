DAA LAB-5, QUESTION 3
Implement Quick Sort of N random elements stored in a file.

-------------------------------------------------------------
OVERALL PLAN
-------------------------------------------------------------
1. GENERATE N random integers and WRITE them to an input file
   ("input.txt"), one per line.
2. READ the N integers back from the file into an array.
3. SORT the array in place using QUICK SORT.
4. WRITE the sorted array to an output file ("output.txt").

-------------------------------------------------------------
ALGORITHM: GENERATE_FILE(filename, N)
-------------------------------------------------------------
1.  open filename for writing
2.  for i <- 1 to N do
3.        write RANDOM(1, 100000) to file, followed by newline
4.  close file

-------------------------------------------------------------
ALGORITHM: READ_FILE(filename) -> A, N
-------------------------------------------------------------
1.  open filename for reading
2.  N <- count of integers in file
3.  allocate array A of size N
4.  for i <- 0 to N-1 do
5.        read A[i] from file
6.  close file
7.  return A, N

-------------------------------------------------------------
ALGORITHM: PARTITION(A, low, high)
-------------------------------------------------------------
1.  pivot <- A[high]                 // Lomuto partition scheme
2.  i <- low - 1
3.  for j <- low to high - 1 do
4.        if A[j] <= pivot then
5.              i <- i + 1
6.              SWAP A[i] with A[j]
7.  SWAP A[i + 1] with A[high]
8.  return i + 1

-------------------------------------------------------------
ALGORITHM: QUICKSORT(A, low, high)
-------------------------------------------------------------
1.  if low < high then
2.        pivotPos <- PARTITION(A, low, high)
3.        QUICKSORT(A, low, pivotPos - 1)
4.        QUICKSORT(A, pivotPos + 1, high)

-------------------------------------------------------------
ALGORITHM: WRITE_FILE(filename, A, N)
-------------------------------------------------------------
1.  open filename for writing
2.  for i <- 0 to N-1 do
3.        write A[i] to file, followed by newline
4.  close file

-------------------------------------------------------------
MAIN DRIVER
-------------------------------------------------------------
1.  GENERATE_FILE("input.txt", N)
2.  A, N <- READ_FILE("input.txt")
3.  QUICKSORT(A, 0, N - 1)
4.  WRITE_FILE("output.txt", A, N)
